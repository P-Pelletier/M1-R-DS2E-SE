get_page = function(urlpage,user_agent){
  res = GET(urlpage, user_agent)
  html = read_html(content(res, "text"))
  clean_html = str_replace_all(as.character(html),'[\\t\\r\\n\\f]','')
  return(clean_html)
}
structure_details = function(split_part){
  details = str_match_all(split_part[-1], '<li>(.*?)</li>')[[1]][, 2]
  return(paste(details,collapse = "\\n"))
}
get_details = function(meal_foodie){
    meal_foodie = meal_foodie[-length(meal_foodie)]
    split_parts = str_split(meal_foodie, '<ul>')
    poles = lapply(split_parts,FUN = function(split_part){str_match(split_part[1], '<li>(.+)')[, 2]})
    details = lapply(split_parts,FUN = structure_details)
    idx = str_detect(poles, "Origine")
    if(any(idx)){
      poles = poles[-which(idx)]
      details = details[-which(idx)]
    }
    poles_details = setNames(details,poles)
    return(poles_details)
}
get_all_meals_for_given_date = function(meal){
  poles_all_infos = list()
  meal_foodies = str_split(meal, '</ul></li>')
  meal_type = str_match(meal, '<div class="meal_title">(.*?)</div>')[, 2]
  for(j in 1:length(meal_foodies)) {
    poles_details  = get_details(meal_foodies[[j]])
    poles_all_infos[[meal_type[[j]]]] = poles_details
  }
  return(poles_all_infos)
}
get_dates_and_meals = function(menus){
  meals = str_split(menus, '<div class="meal">')
  dates = sapply(meals,FUN = function(x){str_match(x[1], '<time class="menu_date_title">Menu du (.*?)</time>')[, 2]})
  clean_meals = lapply(meals,FUN = function(x){x[-1]})
  dates_meals = setNames(clean_meals,dates)
  return(dates_meals)
}
get_all_meals = function(clean_html){
  menus = str_match_all(clean_html,  '<div class="menu">(.*?)</ul></div> *</div>')[[1]][,2]
  dates_meals = get_dates_and_meals(menus)
  for(date in names(dates_meals)) {
    poles_all_infos = get_all_meals_for_given_date(dates_meals[[date]])
    dates_meals[[date]] = poles_all_infos
  }
  return(dates_meals)
}
get_all_links = function(user_agent){
  clean_html = get_page(urlpage = 'https://www.crous-strasbourg.fr/se-restaurer/ou-manger/',
                        user_agent = user_agent)
  links = str_match_all(clean_html, 'href="(.*?)">')[[1]][,2]
  links = links[str_detect(links,'/restaurant/')]
  return(links)
}

get_resto_u_name = function(links){
  resto_names = sapply(str_match_all(links,'restaurant/(.*)/'), FUN = function(x) gsub('-',' ',x[,2]))
  resto_names = toTitleCase(resto_names)
  resto_names = str_replace_all(resto_names,' 2','')
  return(resto_names)
}
